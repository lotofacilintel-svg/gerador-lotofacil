import random from kivy.app import App from kivy.uix.boxlayout import BoxLayout from kivy.uix.gridlayout import GridLayout from kivy.uix.label import Label from kivy.uix.button import Button from kivy.uix.textinput import TextInput from kivy.uix.scrollview import ScrollView from kivy.core.window import Window # Define a cor de fundo (Azul escuro/Grafite moderno) Window.clearcolor = (0.1, 0.12, 0.15, 1) class LotofacilApp(App): def build(self): # Layout Principal root = BoxLayout(orientation='vertical', padding=15, spacing=10) # Título titulo = Label( text='Gerador Lotofácil', font_size='24sp', bold=True, size_hint_y=None, height=50, color=(0.2, 0.8, 0.4, 1) ) root.add_widget(titulo) # Painel de Controles controles = GridLayout(cols=2, spacing=10, size_hint_y=None, height=100) controles.add_widget(Label(text='Qtd. Dezenas (15 a 20):', font_size='16sp')) self.input_dezenas = TextInput( text='15', multiline=False, input_filter='int', font_size='16sp', halign='center' ) controles.add_widget(self.input_dezenas) controles.add_widget(Label(text='Qtd. Jogos:', font_size='16sp')) self.input_jogos = TextInput( text='5', multiline=False, input_filter='int', font_size='16sp', halign='center' ) controles.add_widget(self.input_jogos) root.add_widget(controles) # Botão Gerar btn_gerar = Button( text='GERAR JOGOS', font_size='18sp', bold=True, background_color=(0.2, 0.7, 0.3, 1), size_hint_y=None, height=50 ) btn_gerar.bind(on_press=self.gerar_jogos) root.add_widget(btn_gerar) # Área de Exibição dos Jogos (Com Rolagem) self.scroll = ScrollView(size_hint=(1, 1)) self.lista_resultados = BoxLayout(orientation='vertical', spacing=10, size_hint_y=None) self.lista_resultados.bind(minimum_height=self.lista_resultados.setter('height')) self.scroll.add_widget(self.lista_resultados) root.add_widget(self.scroll) return root def gerar_jogos(self, instance): # Limpa resultados anteriores self.lista_resultados.clear_widgets() # Validação dos campos try: dezenas = int(self.input_dezenas.text) qtd_jogos = int(self.input_jogos.text) except ValueError: self.exibir_mensagem("Por favor, insira números válidos!") return if dezenas < 15 or dezenas > 20: self.exibir_mensagem("A quantidade de dezenas deve ser entre 15 e 20.") return if qtd_jogos <= 0: self.exibir_mensagem("Informe uma quantidade de jogos maior que zero.") return # Lógica de Geração da Lotofácil for i in range(1, qtd_jogos + 1): # Sorteia números de 1 a 25 sem repetição numeros_sorteados = sorted(random.sample(range(1, 26), dezenas)) # Formatando para exibição (ex: 01 - 02 - 05...) texto_numeros = " - ".join(f"{num:02d}" for num: in numeros_sorteados) # Criando o painel visual do jogo card_jogo = BoxLayout(orientation='vertical', size_hint_y=None, height=70, padding=5) lbl_titulo_jogo = Label( text=f"Jogo {i:02d} ({dezenas} dezenas)", font_size='14sp', bold=True, color=(0.9, 0.9, 0.2, 1), size_hint_y=None, height=20 ) lbl_dezenas_jogo = Label( text=texto_numeros, font_size='13sp', color=(1, 1, 1, 1), size_hint_y=None, height=40 ) card_jogo.add_widget(lbl_titulo_jogo) card_jogo.add_widget(lbl_dezenas_jogo) self.lista_resultados.add_widget(card_jogo) def exibir_mensagem(self, texto): msg = Label( text=texto, color=(1, 0.3, 0.3, 1), font_size='15sp', size_hint_y=None, height=40 ) self.lista_resultados.add_widget(msg) if __name__ == '__main__': LotofacilApp().run()
